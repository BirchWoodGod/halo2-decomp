cmake_minimum_required(VERSION 3.20)
project(halo2_engine C)
option(H2_SANITIZE "Enable undefined-behavior checks for native engine code" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(halo2_engine SHARED src/data_array.c src/data_array_alloc.c src/heap.c src/online_tasks.c src/async_tasks.c src/async_task_create.c src/async_task_result.c src/network_observer_async.c src/bitstream.c src/text_codec.c src/session_description.c src/message_dispatch.c src/message_reader.c src/discovery.c src/network_packet.c src/network_send.c src/network_routing.c src/network_cleanup.c src/network_tracking.c src/resource_release.c src/network_observer.c src/network_observer_send.c src/network_resolution.c src/network_storage.c src/network_storage_queue.c src/network_connection.c src/network_registration.c src/network_final_state.c src/network_config.c src/file_path.c src/file_io.c src/host/files.c
    src/engine_pools.c src/xbe_image.c src/crc.c src/hash_table.c src/arena.c src/game_lifecycle.c src/transport.c src/random.c src/network_messages.c src/network_endpoint.c src/network_socket.c src/network_options.c src/network_address.c src/network_bind.c src/network_open.c src/network_state.c src/network_session.c src/network_session_lifecycle.c src/network_session_send.c src/network_session_control.c src/network_session_tasks.c src/network_session_identity.c src/network_observer_admission.c src/network_observer_query.c src/network_observer_retry.c src/network_connection_setup.c src/network_route_insert.c src/network_handshake.c src/network_connection_open.c src/network_observer_selection.c src/network_observer_tick.c src/network_slot_alloc.c src/network_connection_allocate.c src/text_format.c src/network_observer_admit.c src/network_observer_timeout.c src/network_observer_poll.c src/network_observer_bandwidth.c src/network_observer_measurement.c src/network_observer_rates.c src/network_session_membership.c src/network_session_removal.c src/network_parameters.c)
target_include_directories(halo2_engine PUBLIC include PRIVATE src)
target_compile_options(halo2_engine PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(halo2_engine PRIVATE OpenSSL::Crypto m)
add_executable(halo2-engine-host src/host/main.c)
target_include_directories(halo2-engine-host PRIVATE src)
target_link_libraries(halo2-engine-host PRIVATE halo2_engine)
target_compile_options(halo2-engine-host PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(H2_SANITIZE)
    foreach(target halo2_engine halo2-engine-host)
        target_compile_options(${target} PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
        target_link_options(${target} PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
    endforeach()
endif()

set_source_files_properties(src/network_observer_bandwidth.c src/network_observer_measurement.c src/network_observer_rates.c PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
